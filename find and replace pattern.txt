import java.util.*;

class Solution {
    public List<String> findAndReplacePattern(String[] words, String pattern) {
        List<String> result = new ArrayList<>();

        for (String word : words) {
            if (matches(word, pattern)) {
                result.add(word);
            }
        }

        return result;
    }

    private boolean matches(String word, String pattern) {
        // pattern -> word
        char[] map = new char[26];

        // word -> pattern, to ensure bijection
        char[] reverseMap = new char[26];

        Arrays.fill(map, '#');
        Arrays.fill(reverseMap, '#');

        for (int i = 0; i < pattern.length(); i++) {
            char p = pattern.charAt(i);
            char w = word.charAt(i);

            // Pattern character already has a different mapping
            if (map[p - 'a'] != '#' && map[p - 'a'] != w) {
                return false;
            }

            // Word character is already mapped from another pattern character
            if (reverseMap[w - 'a'] != '#' && reverseMap[w - 'a'] != p) {
                return false;
            }

            map[p - 'a'] = w;
            reverseMap[w - 'a'] = p;
        }

        return true;
    }
}
